// ═══════════════════════════════════════════════════════ // CX 온보딩 커리큘럼 - Google Apps Script API // 개인별 체크 현황 관리 버전 // ═══════════════════════════════════════════════════════ function doGet(e) { var action = e.parameter.action || "all"; var callback = e.parameter.callback || ""; var name = e.parameter.name || "unknown"; var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = {}; if (action === "setCheck") { setCheck(ss, name, e.parameter.taskId, e.parameter.checked === "true"); result = { ok: true }; } else if (action === "checks") { result = { checks: getChecks(ss, name) }; } else { if (action === "all" || action === "config") result.config = getConfig(ss); if (action === "all" || action === "tasks") result.tasks = getTasks(ss); if (action === "all" || action === "checks") result.checks = getChecks(ss, name); if (action === "all" || action === "library") result.library = getLibrary(ss); } var json = JSON.stringify(result); var output = callback ? callback + "(" + json + ")" : json; return ContentService .createTextOutput(output) .setMimeType(callback ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } // ── 기본정보: 4행부터 데이터 function getConfig(ss) { var sheet = ss.getSheetByName("기본정보"); if (!sheet) return {}; var data = sheet.getDataRange().getValues(); var config = {}; for (var i = 3; i < data.length; i++) { var key = String(data[i][0] || "").trim(); var val = data[i][1]; if (key && val !== "" && val !== undefined) config[key] = val; } return config; } // ── 일정: 2행 헤더, 3행부터 데이터 function getTasks(ss) { var sheet = ss.getSheetByName("일정"); if (!sheet) return []; var rows = sheet.getDataRange().getValues(); var headers = rows[1]; var tasks = []; for (var i = 2; i < rows.length; i++) { var row = rows[i]; if (!row[0]) continue; var obj = {}; headers.forEach(function(h, idx) { obj[String(h).trim()] = row[idx]; }); tasks.push(obj); } return tasks; } // ── 완료현황: 이름별로 읽기 // 시트 구조: A=항목ID, B=이름, C=완료여부, D=처리시각 function getChecks(ss, name) { var sheet = ss.getSheetByName("완료현황"); if (!sheet) return {}; var rows = sheet.getDataRange().getValues(); var checks = {}; for (var i = 2; i < rows.length; i++) { var taskId = String(rows[i][0] || "").trim(); var rowName = String(rows[i][1] || "").trim(); var checked = rows[i][2]; if (taskId && rowName === name) { checks[taskId] = (checked === true || String(checked).toUpperCase() === "TRUE"); } } return checks; } // ── 자료실: 2행 헤더, 3행부터 데이터 function getLibrary(ss) { var sheet = ss.getSheetByName("자료실"); if (!sheet) return []; var rows = sheet.getDataRange().getValues(); var headers = rows[1]; var libs = []; for (var i = 2; i < rows.length; i++) { var row = rows[i]; if (!row[0]) continue; var obj = {}; headers.forEach(function(h, idx) { obj[String(h).trim()] = row[idx]; }); libs.push(obj); } return libs; } // ── 완료현황 저장: 이름 + 항목ID 기준으로 저장 function setCheck(ss, name, taskId, checked) { var sheet = ss.getSheetByName("완료현황"); if (!sheet) return; var rows = sheet.getDataRange().getValues(); // 같은 이름 + 같은 항목ID 찾기 for (var i = 2; i < rows.length; i++) { var existId = String(rows[i][0] || "").trim(); var existName = String(rows[i][1] || "").trim(); if (existId === taskId && existName === name) { sheet.getRange(i + 1, 3).setValue(checked); sheet.getRange(i + 1, 4).setValue(new Date().toLocaleString("ko-KR")); return; } } // 없으면 새 행 추가 sheet.appendRow([taskId, name, checked, new Date().toLocaleString("ko-KR")]); }
